inferencekey

# Inference-only dependencies for the LP-MusicCaps custom backend.
# The SDK runtime itself imports NONE of these — they belong to this backend.
#
# Modern torch + transformers. BartCaptionModel.generate() was adapted to the
# transformers >= 4.44 generation API (it passes only `encoder_outputs`, no
# manual decoder_input_ids), so the old torch<2.2 / transformers==4.26.1 pins
# are gone. Modern wheels are what let this run on recent GPUs: NVIDIA CUDA and
# AMD ROCm publish wheels for new architectures (e.g. RDNA4/gfx120X) only for
# torch >= 2.7. On an AMD node the worker points pip at the ROCm wheel index so
# `torch` resolves to a `+rocm` build; on NVIDIA it comes from PyPI (CUDA).
# pick_device() then selects the accelerator (falling back to CPU).
torch>=2.2
torchaudio>=2.2
# Upper-bounded to <5: transformers 5.x lazily pulls in torchvision when
# importing BART, and on a ROCm node the base image's torchvision (exposed to
# the venv via the torch-link .pth) is ABI-incompatible with the venv's torch
# ("operator torchvision::nms does not exist"), breaking the BART import. The
# 4.44–4.57 line has the modern generation API this backend targets without the
# torchvision pull-in, and is verified end-to-end on CUDA and ROCm.
transformers>=4.44,<5
librosa>=0.8
numpy<2
einops
soundfile
# audioread lets librosa decode mp3/ogg without a system ffmpeg for common
# cases; ffmpeg on PATH is still the most robust decoder for exotic formats.
audioread
# The model downloads facebook/bart-base from the HF hub. Some nodes preset
# HF_HUB_ENABLE_HF_TRANSFER=1; hf_transfer satisfies it (setup() also disables
# it defensively). Harmless if unused.
hf_transfer
