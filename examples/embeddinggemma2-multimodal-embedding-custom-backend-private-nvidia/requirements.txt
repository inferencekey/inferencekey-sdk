# Requirements of the CUSTOM BACKEND (installed by the worker with uv into the
# backend's own venv). The client side (main.py) only needs the local SDK build
# and `requests`, see README.

# torch from the CUDA 12.6 wheel index: PyPI's default torch 2.14 is the CUDA 13
# build, which needs driver 580 or newer, and many rented GPU hosts are older
# (tested on driver 570 / CUDA 12.8).
--extra-index-url https://download.pytorch.org/whl/cu126
torch==2.14.1
torchvision==0.29.1

# InferenceKey SDK runtime: provides `python -m inferencekey.backend.serve`.
inferencekey

# EmbeddingGemma 2 support landed in transformers 5.19; the card asks for
# sentence-transformers >= 6.1 for ordered multimodal dict inputs.
transformers==5.19.0
sentence-transformers==6.1.0

# Media decoding: Pillow for images, PyAV (FFmpeg bundled) for audio and video.
# No version floors here on purpose: uv only takes a package from the FIRST
# index that has it, and the PyTorch index above also hosts (older) pillow,
# numpy and requests. A floor such as requests>=2.31 makes the whole install
# unsolvable ("only requests==2.28.1 is available").
pillow
av
numpy
requests
