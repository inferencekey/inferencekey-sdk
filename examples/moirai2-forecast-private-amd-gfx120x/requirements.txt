requests>=2.28

# --- The custom backend's deps (installed into the backend's venv) ---------
# The SDK runtime that serves the backend imports NONE of these — they belong
# to backend.py (moirai2). The worker installs them into the backend's venv
# from this file (name = "moirai2", requirements = "requirements.txt").
#
# ROCm/torch note (why this file is unusual): uni2ts pins `torch>=2.1,<2.5`,
# but AMD RDNA4 (gfx120x) ROCm wheels start at `torch>=2.7`. Installing uni2ts
# with its own deps would drag in that `<2.5` pin and make the venv unresolvable
# on this hardware. So we install uni2ts with **--no-deps** and declare its
# inference-time dependencies here ourselves — pinning torch WITHOUT an upper
# bound so the worker's ROCm wheel (2.7+) satisfies it. Moirai's forecast path
# does not use torch APIs that changed across 2.4→2.7.

# The InferenceKey SDK runtime provides `inferencekey.backend.serve`, which the
# worker launches inside the backend venv; keep it resolvable from site-packages.
inferencekey

# Moirai 2.0 (uni2ts) is only published from GitHub. It is installed from a
# branch whose dependency metadata does not cap torch, so the worker's already
# installed ROCm torch (2.7+) is kept. (uni2ts main pins `torch<2.5`, which is
# incompatible with the AMD RDNA4 / gfx120x ROCm wheels that start at 2.7 — a
# requirements.txt cannot pass a per-package `--no-deps` to uv, so the cap is
# avoided at the source instead.) uni2ts's inference-time deps are pinned below.
uni2ts @ git+https://github.com/SalesforceAIResearch/uni2ts.git
